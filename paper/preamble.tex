% ============================================================================
% Claims ladder.  Every displayed key result is wrapped in `keyresult`, whose
% level is machine-extracted by scripts/check_claims.py:
%   I = identity or standard result   C = conditional theorem of the model
%   P = phenomenological / fitted     S = speculative (Outlook only)
% Usage:  \begin{keyresult}{C}{eq:jerk-bound} ... \end{keyresult}
% ============================================================================
% The evidential level is printed with the result.  It used to be discarded,
% so a reader had to match labels against a table in an appendix to learn
% whether a displayed result was an identity, a theorem or a conjecture, and a
% ladder invisible at the point of the claim does not do the job the
% introduction claims for it.  The argument is also extracted from the source
% by scripts/check_claims.py, which greps \begin{keyresult}{LEVEL}{label}.
%
% Built from \csname rather than nested \ifthenelse, and set as a flush-right
% line rather than a \marginpar: the first version of this used both, and
% pdflatex spent thirteen minutes on a single pass without finishing.
\expandafter\def\csname claimlevel@A\endcsname{Postulate}
\expandafter\def\csname claimlevel@I\endcsname{Identity}
\expandafter\def\csname claimlevel@C\endcsname{Theorem of the model}
\expandafter\def\csname claimlevel@X\endcsname{Result of an auxiliary model}
\expandafter\def\csname claimlevel@N\endcsname{Numerical}
\expandafter\def\csname claimlevel@P\endcsname{Fitted to data}
\expandafter\def\csname claimlevel@S\endcsname{Conjecture}
\newcommand{\claimlevelname}[1]{%
  \ifcsname claimlevel@#1\endcsname\csname claimlevel@#1\endcsname\else#1\fi}
\newenvironment{keyresult}[2]%
  {\par\nobreak\vspace{-2pt}%
   {\raggedleft\scriptsize\sffamily\color{black!55}%
    #1 --- \claimlevelname{#1}\par}%
   \nobreak\vspace{-6pt}%
   \begin{equation}\label{#2}\begin{aligned}}%
  {\end{aligned}\end{equation}}
